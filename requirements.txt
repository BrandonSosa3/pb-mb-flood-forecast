requests      # download data from web APIs (NOAA, CDIP)
pandas        # tables of time + water level, math on columns
matplotlib    # quick charts while exploring
xarray        # read gridded/scientific data files (CDIP wave model)
netCDF4       # the file format CDIP and NOAA use (.nc), read over the web
rasterio      # read/write map grids (GeoTIFF elevation files)
scipy         # image tools: find connected areas (water, flood spreading)
scikit-image  # fast 'how high must water rise to reach here' calculation (morphological reconstruction)
geopandas     # tables of map shapes (buildings, streets) with geometry
shapely       # geometry math: lengths, buffers, splitting lines
laspy[lazrs]  # read lidar point clouds (.laz) for building heights (3D view)
numba         # fast loops for terrain hydrology on 100M+ cells (rain_terrain.py)
